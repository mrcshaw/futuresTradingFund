//@version=5
// How this script works
// The script’s entry is a Bollinger Band mean reversion. A long opens when price closes below the lower band. A short opens when price closes above the upper band.
// The stop is a fixed dollar stop, default 250, converted to price points by the point value.
strategy("Bollinger Mean Reversion", overlay=true,
  initial_capital=50000, default_qty_type=strategy.fixed, default_qty_value=1,
  commission_type=strategy.commission.cash_per_contract, commission_value=5.50,
  slippage=1, pyramiding=0, process_orders_on_close=true, calc_on_every_tick=false,
  max_bars_back=5000, max_boxes_count=500)
// ===== RISK =====
pv                = input.float(50, "Point Value ($)")
stop_dollars      = input.int(250, "Stop ($)", minval=250, maxval=1500)
tp_dollars        = input.int(500, "Take Profit ($)", minval=250, maxval=2000)
use_trail         = input.bool(false, "Enable Trailing Stop")
trail_pct         = input.float(50, "Trail keep (%)", minval=0, maxval=100)
trail_step_amt    = input.int(500, "Trail: Tighten every ($)", minval=0, maxval=2000)
trail_step_pct    = input.float(50, "Trail step (%)", minval=0, maxval=100)
trail_min_dollars = input.int(1000, "Trail: Min Profit ($)", minval=0, maxval=2000)
long_stop = close - stop_dollars / pv
long_limit = close + tp_dollars / pv
short_stop = close + stop_dollars / pv
short_limit = close - tp_dollars / pv
if strategy.position_size != 0 and not na(stop_dollars)
    if strategy.openprofit <= -stop_dollars
        strategy.close_all(comment="Open P&L > SL Risk")
// ===== SESSION =====
use_session   = input.bool(true, "Use Session Filter")
sess_start_hr = input.int(9, "Session Start Hour (ET)")
sess_start_mn = input.int(30,  "Session Start Minute")
sess_end_hr   = input.int(16,  "Session End Hour (ET)")
sess_end_mn   = input.int(0,   "Session End Minute")
sess_bar_hr = hour(time, "America/New_York")
sess_bar_mn = minute(time, "America/New_York")
sess_bar_t = sess_bar_hr * 100 + sess_bar_mn
sess_clock_start = sess_start_hr * 100 + sess_start_mn
sess_clock_end = sess_end_hr * 100 + sess_end_mn
in_session = not use_session or (sess_clock_start > sess_clock_end ? (sess_bar_t >= sess_clock_start or sess_bar_t < sess_clock_end) : (sess_bar_t >= sess_clock_start and sess_bar_t < sess_clock_end))
if not in_session
    strategy.close_all(comment="Session")

// ===== ENTRY =====
bb_len = 20
bb_mult = 2.0
[bb_mid, bb_up, bb_low] = ta.bb(close, bb_len, bb_mult)
long_sig = close < bb_low
short_sig = close > bb_up

if long_sig
    if in_session
        strategy.entry("Long", strategy.long)
    strategy.exit("Long TP/SL", "Long", limit=long_limit, stop=long_stop)

if short_sig
    if in_session
        strategy.entry("Short", strategy.short)
    strategy.exit("Short TP/SL", "Short", limit=short_limit, stop=short_stop)

plot(bb_mid, "BB Mid", color.gray)
plot(bb_up, "BB Upper", color.red)
plot(bb_low, "BB Lower", color.green)
