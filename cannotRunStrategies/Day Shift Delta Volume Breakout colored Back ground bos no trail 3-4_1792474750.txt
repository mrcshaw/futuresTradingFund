//@version=5
strategy("Day Shift Delta Volume Breakout", overlay=true,
  initial_capital=50000, default_qty_type=strategy.fixed, default_qty_value=1,
  commission_type=strategy.commission.cash_per_contract, commission_value=5.50,
  slippage=1, pyramiding=0, process_orders_on_close=true, calc_on_every_tick=false,
  max_bars_back=5000)
// ===== RISK =====
pv                = input.float(50, "Point Value ($)")
stop_dollars      = input.int(1500, "Stop ($)", minval=250, maxval=1500)
tp_dollars        = input.int(1750, "Take Profit ($)", minval=250, maxval=2000)
use_trail         = input.bool(false, "Enable Trailing Stop")
trail_step_amt    = input.int(500, "Trail: Tighten every ($)", minval=0, maxval=2000)
trail_min_dollars = input.int(1000, "Trail: Min Profit ($)", minval=0, maxval=2000)
long_stop = close - stop_dollars / pv
long_limit = close + tp_dollars / pv
short_stop = close + stop_dollars / pv
short_limit = close - tp_dollars / pv
if strategy.position_size != 0 and not na(stop_dollars)
    if strategy.openprofit <= -stop_dollars
        strategy.close_all(comment="Open P&L > SL Risk")
trail_step_pct = input.float(50, "trail_step_pct")
trail_pct = input.float(50, "trail_pct")

// ===== VOLUME PROFILE =====
profile_lookback = input.int(120, "Profile Lookback (bars)", minval=20, maxval=2000, step=10)
profile_rows     = input.int(10, "Price Levels (rows)", minval=5, maxval=100, step=5, tooltip="More rows = finer resolution but slower.")

// ===== DELTA VOLUME =====
min_delta_pct = input.float(5.0, "Min Delta Imbalance (%)", minval=0, maxval=80, step=5, tooltip="Net delta at POC must exceed this % of total POC volume to confirm direction. 0 = disabled.")

// ===== TREND FILTER =====
vwma_len       = input.int(50, "Trend VWMA Length", minval=1)
vwma_slope_len = input.int(5, "VWMA Slope Lookback (bars)", minval=1)
min_slope      = input.float(0.0, "Min VWMA Slope (pts)", minval=0.0, tooltip="0 = disabled.")

// ===== ENTRY =====
fractal_len   = input.int(5, "Fractal Length (BOS)", minval=3, maxval=21, step=2, tooltip="LuxAlgo fractal length for BOS detection.")
cooldown_bars = input.int(15, "Cooldown (bars)", minval=0, maxval=100, step=1)

// ===== FILTERS (Distractors) =====
use_price_vwma  = input.bool(true, "Price vs VWMA", group="Filters", tooltip="Long: close above VWMA. Short: close below VWMA.")
use_bos_candle  = input.bool(true, "BOS Bar Color", group="Filters", tooltip="BOS bar must be same color as direction (bullish for long, bearish for short).")
use_vol_filter  = input.bool(true, "High Volume on BOS", group="Filters", tooltip="BOS bar must have above-average volume.")
vol_ma_len     = input.int(50, "Volume MA Length", minval=5, maxval=100, group="Filters")

// ===== SESSION =====
use_session   = input.bool(true, "Use Session Filter")
sess_start_hr = input.int(8, "Session Start Hour (ET)")
sess_start_mn = input.int(0,  "Session Start Minute")
sess_end_hr   = input.int(16,  "Session End Hour (ET)")
sess_end_mn   = input.int(0,  "Session End Minute")

// ===== CALCULATIONS =====
// Volume Profile POC Calculation
var float poc_price = na
var float poc_volume = 0.0
var float poc_hi = na
var float poc_lo = na
var float poc_delta = 0.0

if bar_index >= profile_lookback
    float high_range = high[profile_lookback-1]
    float low_range = low[profile_lookback-1]
    float range_size = high_range - low_range
    if range_size > 0
        float row_height = range_size / profile_rows
        float[] row_volumes = array.new_float(profile_rows, 0.0)
        float[] row_deltas = array.new_float(profile_rows, 0.0)
        
        for i = 0 to profile_lookback - 1
            float bar_high = high[i]
            float bar_low = low[i]
            float bar_close = close[i]
            float bar_vol = volume[i]
            
            // Determine which rows this bar touches
            int start_row = math.max(0, math.floor((bar_low - low_range) / row_height))
            int end_row = math.min(profile_rows - 1, math.floor((bar_high - low_range) / row_height))
            
            for r = start_row to end_row
                float row_low = low_range + r * row_height
                float row_high = row_low + row_height
                
                // Calculate volume portion in this row
                float overlap_low = math.max(bar_low, row_low)
                float overlap_high = math.min(bar_high, row_high)
                float overlap_size = overlap_high - overlap_low
                float bar_size = bar_high - bar_low
                float vol_portion = bar_size > 0 ? bar_vol * (overlap_size / bar_size) : 0.0
                
                // Determine delta direction (simplified: close vs open)
                float delta_dir = close[i] >= open[i] ? 1.0 : -1.0
                float delta_portion = vol_portion * delta_dir
                
                array.set(row_volumes, r, array.get(row_volumes, r) + vol_portion)
                array.set(row_deltas, r, array.get(row_deltas, r) + delta_portion)
        
        // Find POC (Max Volume Row)
        int max_idx = 0
        float max_vol = array.get(row_volumes, 0)
        for r = 1 to profile_rows - 1
            if array.get(row_volumes, r) > max_vol
                max_vol := array.get(row_volumes, r)
                max_idx := r
        
        poc_price := low_range + (max_idx + 0.5) * row_height
        poc_volume := max_vol
        poc_hi := low_range + (max_idx + 1) * row_height
        poc_lo := low_range + max_idx * row_height
        poc_delta := array.get(row_deltas, max_idx)

// VWMA Calculation
vwma_val = ta.vwma(close, vwma_len)
vwma_up = vwma_val > vwma_val[1]
vwma_dn = vwma_val < vwma_val[1]

// Fractal / BOS Detection
// Simplified Fractal: High is higher than n bars left and right
is_high_fractal = high > high[1] and high > high[2] and high > high[3] and high > high[4] and high > high[1] and high > high[2] and high > high[3] and high > high[4]
is_low_fractal = low < low[1] and low < low[2] and low < low[3] and low < low[4] and low < low[1] and low < low[2] and low < low[3] and low < low[4]

var float last_high_fractal = na
var float last_low_fractal = na

if is_high_fractal
    last_high_fractal := high
if is_low_fractal
    last_low_fractal := low

// Break of Structure (BOS)
long_bos = not na(last_high_fractal) and close > last_high_fractal
short_bos = not na(last_low_fractal) and close < last_low_fractal

// Filters
vol_ma = ta.sma(volume, vol_ma_len)
high_vol = volume > vol_ma
bullish_candle = close > open
bearish_candle = close < open

price_above_vwma = close > vwma_val
price_below_vwma = close < vwma_val

// Session Filter
in_session = time(timeframe.period, "0800-1600", "America/New_York")

// Entry Logic
// Relaxed entry to increase trade count: 
// 1. Must be in session
// 2. Must have a BOS
// 3. Must align with VWMA trend
// 4. Must have volume confirmation
// 5. Delta imbalance check is kept but we allow it to be less strict or just use it as a filter if present
// To increase trades, we will remove the strict POC delta requirement from the entry trigger 
// and instead rely on the BOS + Trend + Volume. The POC is still plotted but not a hard gate for entry 
// unless the delta is strongly confirming. Given the low trade count, removing the POC delta gate is the primary change.

long_sig = in_session and long_bos and price_above_vwma and high_vol and bullish_candle
short_sig = in_session and short_bos and price_below_vwma and high_vol and bearish_candle

// Cooldown
var int last_trade_bar = -1
if strategy.position_size != 0
    last_trade_bar := bar_index

can_trade = bar_index - last_trade_bar >= cooldown_bars

if can_trade
    if long_sig and strategy.position_size == 0
        strategy.entry("Long", strategy.long)
    else if short_sig and strategy.position_size == 0
        strategy.entry("Short", strategy.short)

// Exit Logic
if strategy.position_size > 0
    strategy.exit("XL", "Long", stop=long_stop, limit=long_limit)
else if strategy.position_size < 0
    strategy.exit("XS", "Short", stop=short_stop, limit=short_limit)

// Flatten at 16:45
if time(timeframe.period, "1645-1646", "America/New_York") and strategy.position_size != 0
    strategy.close_all(comment="EOD Flatten")

// ===== PLOT =====
plot(vwma_val, "VWMA", vwma_up ? color.lime : vwma_dn ? color.red : color.gray, linewidth=2)
plot(not na(poc_price) ? poc_price : na, "POC", color.yellow, style=plot.style_stepline, linewidth=1)
plot(not na(poc_hi) ? poc_hi : na, "POC High", #2962FF, style=plot.style_stepline, linewidth=2)
plot(not na(poc_lo) ? poc_lo : na, "POC Low", #2962FF, style=plot.style_stepline, linewidth=2)

plotshape(long_sig, "Long Signal", shape.triangleup, location.belowbar, color.lime, size=size.small)
plotshape(short_sig, "Short Signal", shape.triangledown, location.abovebar, color.red, size=size.small)
