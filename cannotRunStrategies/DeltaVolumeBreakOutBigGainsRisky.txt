//@version=5
// How this script works
// desk indicator ta.rsi
_desk_gate = ta.rsi(close, 14)
_desk_long = _desk_gate > 50
_desk_short = _desk_gate < 50
// The code uses strategy.entry("Long", strategy.long) and strategy.entry("Short", strategy.short) entries after a pullback-to-EMA setup.
last_trade_bar := bar_index
// A long requires the session condition, price above the EMA, a positive EMA slope, a bar low that touches the EMA while closing above it, cooldown, no open position, and RSI(14) above 50.
// A short requires the session condition, price below the EMA, a negative EMA slope, a bar high that touches the EMA while closing below it, cooldown, no open position, and RSI(14) below 50.
// The long stop is long_stop, set at close minus stop_dollars divided by pv, and the short stop is short_stop, set at close plus stop_dollars divided by pv.
// It also exits at a take-profit limit, closes if open P&L reaches the stop dollar amount, and can flatten at 16:45 New York.

strategy("Delta Volume Breakout", overlay=true,
  initial_capital=50000, default_qty_type=strategy.fixed, default_qty_value=1,
  commission_type=strategy.commission.cash_per_contract, commission_value=5.50,
  slippage=1, pyramiding=0, process_orders_on_close=true, calc_on_every_tick=false,
  max_bars_back=5000)
// ===== RISK =====
pv                = input.float(50, "Point Value ($)")
stop_dollars      = input.int(1500, "Stop ($)", minval=250, maxval=1500)
tp_dollars        = input.int(1750, "Take Profit ($)", minval=250, maxval=2000)
use_trail         = input.bool(false, "Enable Trailing Stop")
trail_pct         = input.float(50, "Trail keep (%)", minval=0, maxval=100)
trail_step_amt    = input.int(500, "Trail: Tighten every ($)", minval=0, maxval=2000)
trail_step_pct    = input.float(50, "Trail step (%)", minval=0, maxval=100)
trail_min_dollars = input.int(1000, "Trail: Min Profit ($)", minval=0, maxval=2000)
long_stop = close - stop_dollars / pv
long_limit = close + tp_dollars / pv
short_stop = close + stop_dollars / pv
short_limit = close - tp_dollars / pv
if strategy.position_size != 0 and not na(stop_dollars)
    if strategy.openprofit <= -stop_dollars
        strategy.close_all(comment="Open P&L > SL Risk")

// ===== TREND FILTER =====
ema_len       = input.int(20, "Trend EMA Length", minval=1)
ema_slope_len = input.int(5, "EMA Slope Lookback (bars)", minval=1)
min_slope     = input.float(0.1, "Min EMA Slope (pts)", minval=0.0, tooltip="0 = disabled.")

// ===== ENTRY =====
confirm_candles = input.int(3, "Confirmation Candles", minval=1, maxval=10, step=1, tooltip="Number of consecutive same-color candles in the trade direction required to confirm entry.")
cooldown_bars   = input.int(10, "Cooldown (bars)", minval=0, maxval=100, step=1)

// ===== SESSION =====
use_session   = input.bool(true, "Use Session Filter")
sess_start_hr = input.int(20, "Session Start Hour (ET)")
sess_start_mn = input.int(0,  "Session Start Minute")
sess_end_hr   = input.int(16,  "Session End Hour (ET)")
sess_end_mn   = input.int(0,  "Session End Minute")

// ===== COMPUTED =====
float stop_pts = stop_dollars / pv
float tp_pts   = tp_dollars / pv

// ===== SESSION LOGIC =====
bar_hr = hour(time, "America/New_York")
bar_mn = minute(time, "America/New_York")
bar_t  = bar_hr * 100 + bar_mn
int s_start = sess_start_hr * 100 + sess_start_mn
int s_end   = sess_end_hr * 100 + sess_end_mn

bool in_session = not use_session or (s_start > s_end ? (bar_t >= s_start or bar_t < s_end) : (bar_t >= s_start and bar_t < s_end))
bool flatten_time = in_session and (bar_t >= 1645)

// ===== INDICATORS =====
ema_val = ta.ema(close, ema_len)
ema_prev = ta.valuewhen(ema_val != ta.ema(close, ema_len), ema_val, 1)

// New Entry Logic: Pullback to EMA in Trend
// Long: Price above EMA, EMA sloping up, price pulls back to touch EMA, then closes back above
// Short: Price below EMA, EMA sloping down, price pulls back to touch EMA, then closes back below

ema_slope = ta.change(ema_val, ema_slope_len)
bool trend_up = close > ema_val and ema_slope > min_slope
bool trend_down = close < ema_val and ema_slope < -min_slope

// Pullback condition: Low touches EMA (for long) or High touches EMA (for short)
bool pullback_long = low <= ema_val and close > ema_val
bool pullback_short = high >= ema_val and close < ema_val

// Cooldown Logic
var int last_trade_bar = na
bool cooldown_ok = na(last_trade_bar) or (bar_index - last_trade_bar >= cooldown_bars)

// Entry Signals
long_sig = in_session and trend_up and pullback_long and cooldown_ok and strategy.position_size == 0
short_sig = in_session and trend_down and pullback_short and cooldown_ok and strategy.position_size == 0

if long_sig and _desk_long
    strategy.entry("Long", strategy.long)
    last_trade_bar := bar_index
if short_sig and _desk_short
    strategy.entry("Short", strategy.short)
    last_trade_bar := bar_index

// ===== EXIT LOGIC =====
if strategy.position_size > 0
    strategy.exit("XL", "Long", stop=long_stop, limit=long_limit)
else if strategy.position_size < 0
    strategy.exit("XS", "Short", stop=short_stop, limit=short_limit)

// Flatten at 16:45 New York
if flatten_time and strategy.position_size != 0
    strategy.close_all(comment="EOD Flatten")

// ===== PLOT =====
plot(ema_val, "Trend EMA", close > ema_val ? color.lime : color.red, linewidth=2)
plotshape(long_sig, "Long Signal", shape.triangleup, location.belowbar, color.lime, size=size.small)
plotshape(short_sig, "Short Signal", shape.triangledown, location.abovebar, color.red, size=size.small)
