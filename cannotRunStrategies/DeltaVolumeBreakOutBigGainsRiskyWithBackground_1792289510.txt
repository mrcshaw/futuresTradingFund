//@version=5
// How this script works
// The entry is a Mean Reversion, with price crossing below the 20-period Simple Moving Average for longs and above it for shorts.
// The cross must also pass the session filter, cooldown, and flat-position checks.
// The stop is a dollar stop, default 250, converted to points using the point value.
// The script also uses a take-profit order, an end-of-day close, and an open P&L check that flattens if the loss exceeds the stop risk.

strategy("Mean Reversion with background", overlay=true,
  initial_capital=50000, default_qty_type=strategy.fixed, default_qty_value=1,
  commission_type=strategy.commission.cash_per_contract, commission_value=5.50,
  slippage=1, pyramiding=0, process_orders_on_close=false, calc_on_every_tick=true,
  max_bars_back=5000)
// ===== RISK =====
pv                = input.float(20, "Point Value ($)")
stop_dollars      = input.int(250, "Stop ($)", minval=250, maxval=1500)
tp_dollars        = input.int(500, "Take Profit ($)", minval=250, maxval=2000)
use_trail         = input.bool(false, "Enable Trailing Stop")
trail_step_amt    = input.int(500, "Trail: Tighten every ($)", minval=0, maxval=2000)
trail_min_dollars = input.int(1000, "Trail: Min Profit ($)", minval=0, maxval=2000)
trail_pct         = input.float(50, "Trail: Keep % of Peak Profit")
trail_step_pct    = input.float(50, "Trail: Tighten by X% per step")
long_stop = close - stop_dollars / pv
long_limit = close + tp_dollars / pv
short_stop = close + stop_dollars / pv
short_limit = close - tp_dollars / pv
if strategy.position_size != 0 and not na(stop_dollars)
    if strategy.openprofit <= -stop_dollars
        strategy.close_all(comment="Open P&L > SL Risk")

// ===== MEAN REVERSION FILTER =====
mr_len = 15
sma_line = ta.sma(close, mr_len)

// ===== ENTRY =====
confirm_candles = input.int(2, "Confirmation Candles", minval=1, maxval=10, step=1, tooltip="Number of consecutive same-color candles in the trade direction required to confirm entry.")
cooldown_bars   = input.int(40, "Cooldown (bars)", minval=0, maxval=100, step=1)

// ===== SESSION =====
use_session   = input.bool(true, "Use Session Filter")
sess_start_hr = input.int(18, "Session Start Hour (ET)")
sess_start_mn = input.int(0,  "Session Start Minute")
sess_end_hr   = input.int(8,  "Session End Hour (ET)")
sess_end_mn   = input.int(0,  "Session End Minute")

// Session Filter
bool in_session = true
if use_session
    int h = hour(time, "America/New_York")
    int m = minute(time, "America/New_York")
    int start = sess_start_hr * 60 + sess_start_mn
    int end = sess_end_hr * 60 + sess_end_mn
    int now = h * 60 + m
    if start < end
        in_session := now >= start and now < end
    else
        in_session := now >= start or now < end

// Cooldown
var int last_trade_bar = -1000
bool cooldown_ok = bar_index - last_trade_bar > cooldown_bars

// Entry Logic: Mean Reversion
bool long_sig = false
bool short_sig = false

if in_session and cooldown_ok and strategy.position_size == 0
    // Long: Price crosses below SMA (oversold bounce)
    if ta.crossunder(close, sma_line)
        long_sig := true
    // Short: Price crosses above SMA (overbought pullback)
    if ta.crossover(close, sma_line)
        short_sig := true

// Execute Trades
if long_sig
    strategy.entry("L", strategy.long)
    last_trade_bar := bar_index
if short_sig
    strategy.entry("S", strategy.short)
    last_trade_bar := bar_index

// Exit Logic
if strategy.position_size > 0
    strategy.exit("XL", "Long", stop=long_stop, limit=long_limit)
if strategy.position_size < 0
    strategy.exit("XS", "Short", stop=short_stop, limit=short_limit)

// Flatten at 16:45 New York
if time(timeframe.period, "1645-1645", "America/New_York")
    strategy.close_all(comment="EOD")

// Plots
plot(sma_line, "SMA", color.blue, linewidth=2)
plotshape(long_sig, "Long Signal", shape.triangleup, location.belowbar, color.lime, size=size.small)
plotshape(short_sig, "Short Signal", shape.triangledown, location.abovebar, color.red, size=size.small)
