//@version=5
strategy("Post-Open POC (ORB v5)", overlay=true,
  initial_capital=50000, default_qty_type=strategy.fixed, default_qty_value=1,
  commission_type=strategy.commission.cash_per_contract, commission_value=5.50,
  slippage=1, pyramiding=0, process_orders_on_close=false, calc_on_every_tick=true,
  max_bars_back=5000)
// ===== RISK =====
pv                = input.float(50, "Point Value ($)")
stop_dollars      = input.int(250, "Stop ($)", minval=50, maxval=700)
tp_dollars        = input.int(500, "Take Profit ($)", minval=50)
use_trail         = input.bool(false, "Enable Trailing Stop")
trail_pct         = input.float(50, "Trail keep (%)", minval=0, maxval=100)
trail_step_amt    = input.int(500, "Trail: Tighten every ($)", minval=0, maxval=2000)
trail_step_pct    = input.float(50, "Trail step (%)", minval=0, maxval=100)
trail_min_dollars = input.int(1000, "Trail: Min Profit ($)", minval=0, maxval=2000)
long_stop = strategy.position_size > 0 ? strategy.position_avg_price - stop_dollars / pv : close - stop_dollars / pv
long_limit = close + tp_dollars / pv
short_stop = strategy.position_size < 0 ? strategy.position_avg_price + stop_dollars / pv : close + stop_dollars / pv
short_limit = close - tp_dollars / pv
if strategy.position_size != 0 and not na(stop_dollars)
    if strategy.openprofit <= -stop_dollars
        strategy.close_all(comment="Open P&L > SL Risk")
// ===== SESSION =====
use_session   = input.bool(true, "Use Session Filter")
sess_start_hr = input.int(18, "Session Start Hour (ET)")
sess_start_mn = input.int(0,  "Session Start Minute")
sess_end_hr   = input.int(16,  "Session End Hour (ET)")
sess_end_mn   = input.int(0,  "Session End Minute")
sess_bar_hr = hour(time, "America/New_York")
sess_bar_mn = minute(time, "America/New_York")
sess_bar_t = sess_bar_hr * 100 + sess_bar_mn
sess_clock_start = sess_start_hr * 100 + sess_start_mn
sess_clock_end = sess_end_hr * 100 + sess_end_mn
in_session = not use_session or (sess_clock_start > sess_clock_end ? (sess_bar_t >= sess_clock_start or sess_bar_t < sess_clock_end) : (sess_bar_t >= sess_clock_start and sess_bar_t < sess_clock_end))
if not in_session
    strategy.close_all(comment="Session")

// ===== MODE PRESETS =====
mode = input.string("Profit", "Mode", options=["Profit","Balanced","Safe"],
  tooltip="Profit: m1.0 no OR filter | 82% win streak≤1 $12k/60d.\nBalanced: m1.0 OR 2-30 | 81% win streak≤1 $8.4k.\nSafe: m0.5 OR 2-20 | 100% win 0 losses $4.5k.")

// ===== TIMING =====
gather_min   = input.int(15, "Gather Minutes", minval=5, maxval=60, step=5,
  tooltip="Minutes after 9:30 to build opening range. 15 = optimal.")
entry_end_hr = input.int(11, "Entry End Hour", minval=10, maxval=14)
entry_end_mn = input.int(0, "Entry End Minute", minval=0, maxval=59, step=5)
eod_hr       = input.int(15, "EOD Flatten Hour")
eod_mn       = input.int(45, "EOD Flatten Min")

i_tp         = input.float(500.0, "Take Profit ($)", tooltip="Overridden by mode preset")
i_trail_act  = input.float(400.0, "Trail Activate ($)")
i_trail_pct  = input.float(40.0, "Trail Keep %")
max_stop     = input.float(500.0, "Max Stop Cap ($)", minval=0, step=50,
  tooltip="Only cap when OR stop would exceed this. 0=no cap. 500=cap wide-OR days only, let runners breathe.")

// ===== FILTERS =====
i_margin     = input.float(1.0, "Breakout Margin (pts)", minval=0, step=0.25,
  tooltip="Close must be this many pts beyond OR boundary. 1.0 = best. Eliminates fake breakouts.")
i_or_min     = input.float(0.0, "OR Min Size (pts)", minval=0, step=1,
  tooltip="Skip days with opening range < this. 2 for Safe/Balanced.")
i_or_max     = input.float(9999.0, "OR Max Size (pts)", minval=1, step=5,
  tooltip="Skip days with opening range > this. 20-30 filters choppy days.")
i_poc_filter = input.bool(false, "POC Directional Filter")
max_attempts = input.int(2, "Max Attempts / Day", minval=1, maxval=5)

// Apply mode presets
float trail_act_d   = mode == "Profit" ? 400.0 : mode == "Balanced" ? 400.0 : 400.0
float margin_pts    = mode == "Profit" ? 1.0   : mode == "Balanced" ? 1.0   : 0.5
float or_min_pts    = mode == "Profit" ? 0.0   : mode == "Balanced" ? 2.0   : 2.0
float or_max_pts    = mode == "Profit" ? 9999.0: mode == "Balanced" ? 30.0  : 20.0
bool  poc_filter    = false

float tp_pts        = tp_dollars / pv
float trail_act_pts = trail_act_d / pv
float max_stop_pts  = max_stop / pv

// ===== PREV DAY POC (pivot approximation) =====
[prev_hi, prev_lo, prev_cl] = request.security(syminfo.tickerid, "D",
  [high[1], low[1], close[1]], lookahead=barmerge.lookahead_off)
float prev_poc = (prev_hi + prev_lo + prev_cl) / 3.0

// ===== TIME HELPERS =====
is_new_day = ta.change(time("D")) != 0
t = time("1", "0930-1600", "America/New_York")
in_rth = not na(t)

hrs = hour(time, "America/New_York")
mns = minute(time, "America/New_York")
float mins_from_open = (hrs - 9) * 60.0 + (mns - 30)
bool in_gather = in_rth and mins_from_open >= 0 and mins_from_open < gather_min
bool gather_done = in_rth and mins_from_open >= gather_min
int bar_time = hrs * 100 + mns
bool in_entry_window = gather_done and bar_time < (entry_end_hr * 100 + entry_end_mn)
bool at_eod = bar_time >= (eod_hr * 100 + eod_mn)

// ===== OPENING RANGE =====
var float or_high = na
var float or_low  = na
var bool  or_built = false
var bool  or_valid = false

if is_new_day
    or_high  := na
    or_low   := na
    or_built := false
    or_valid := false

if in_gather
    or_high := na(or_high) ? high : math.max(or_high, high)
    or_low  := na(or_low)  ? low  : math.min(or_low, low)

if gather_done and not or_built
    or_built := true
    float or_range = nz(or_high) - nz(or_low)
    or_valid := or_range >= or_min_pts and or_range <= or_max_pts and or_range > 0

// ===== ATTEMPT COUNTER =====
var int day_attempts = 0
if is_new_day
    day_attempts := 0

// ===== TRAILING STOP STATE =====
var float peak_profit     = 0.0
var float current_stop   = na
var float plot_stop      = na
var float entry_stop_px  = na  // Original stop at entry — never moves, checked every tick
var bool  close_pending  = false  // Prevent repeated strategy.close() on every tick

if is_new_day
    peak_profit    := 0.0
    current_stop   := na
    plot_stop      := na
    entry_stop_px  := na
    close_pending  := false

// ===== EOD FLATTEN =====
if at_eod and strategy.position_size != 0
    strategy.close_all(comment="EOD")
    peak_profit    := 0.0
    current_stop   := na
    entry_stop_px  := na
    close_pending  := false

// ===== ENTRY SIGNALS =====
bool can_trade = in_entry_window and or_built and or_valid and day_attempts < max_attempts

// Breakout with margin: close must be margin_pts beyond OR boundary
bool breakout_long  = can_trade and close > (or_high + margin_pts)
bool breakout_short = can_trade and close < (or_low - margin_pts)

// POC directional filter
if poc_filter and not na(prev_poc)
    if close <= prev_poc
        breakout_long := false
    if close >= prev_poc
        breakout_short := false

// Only enter when flat — no reversals.
bool long_sig  = breakout_long and strategy.position_size == 0
bool short_sig = breakout_short and strategy.position_size == 0

// ===== ENTRY + EXIT =====
// Entries: bar close only (barstate.isconfirmed)
// Exits: check every tick (calc_on_every_tick). Record original stop at entry; exit when price crosses it.
bool entry_ok = barstate.isconfirmed

if long_sig and entry_ok
    day_attempts += 1
    peak_profit   := 0.0
    current_stop  := or_low
    close_pending := false
    // Record original stop at entry — never changes
    entry_stop_px := (max_stop_pts > 0 and (close - or_low) > max_stop_pts) ? (close - max_stop_pts) : or_low
    if in_session
        strategy.entry("Long", strategy.long)
else if short_sig and entry_ok
    day_attempts += 1
    peak_profit   := 0.0
    current_stop  := or_high
    close_pending := false
    // Record original stop at entry — never changes
    entry_stop_px := (max_stop_pts > 0 and (or_high - close) > max_stop_pts) ? (close + max_stop_pts) : or_high
    if in_session
        strategy.entry("Short", strategy.short)
else if strategy.position_size > 0
    float avg = strategy.position_avg_price
    float tp_price = avg + tp_pts
    float profit_now = high - avg
    if profit_now > peak_profit
        peak_profit := profit_now
    if peak_profit >= trail_act_pts
        float lock = peak_profit * (trail_pct / 100.0)
        float ns = avg + lock
        if ns > current_stop
            current_stop := ns
    // Exit when price crosses stop. Original stop (entry_stop_px) is floor; trail can raise it.
    // Only call strategy.close() once per stop-out — not on every tick.
    float eff_stop = math.max(nz(entry_stop_px), current_stop)
    plot_stop := eff_stop
    if not close_pending
        if low <= eff_stop
            close_pending := true
            strategy.close("Long", comment="SL")
        else if high >= tp_price
            close_pending := true
            strategy.close("Long", comment="TP")
else if strategy.position_size < 0
    float avg = strategy.position_avg_price
    float tp_price = avg - tp_pts
    float profit_now = avg - low
    if profit_now > peak_profit
        peak_profit := profit_now
    if peak_profit >= trail_act_pts
        float lock = peak_profit * (trail_pct / 100.0)
        float ns = avg - lock
        if ns < current_stop
            current_stop := ns
    // Exit when price crosses stop. Original stop (entry_stop_px) is ceiling; trail can lower it.
    // Only call strategy.close() once per stop-out — not on every tick.
    float eff_stop = math.min(nz(entry_stop_px), current_stop)
    plot_stop := eff_stop
    if not close_pending
        if high >= eff_stop
            close_pending := true
            strategy.close("Short", comment="SL")
        else if low <= tp_price
            close_pending := true
            strategy.close("Short", comment="TP")

// ===== RESET ON FLAT =====
if strategy.position_size == 0 and strategy.position_size[1] != 0
    peak_profit   := 0.0
    current_stop  := na
    plot_stop     := na
    entry_stop_px := na
    close_pending := false

// ===== PLOTS =====
color or_hi_col = or_built and or_valid ? color.new(color.lime, 20) : or_built ? color.new(color.gray, 60) : na
color or_lo_col = or_built and or_valid ? color.new(color.red, 20) : or_built ? color.new(color.gray, 60) : na

plot(or_built ? or_high : na, "OR High", or_hi_col, linewidth=2, style=plot.style_stepline)
plot(or_built ? or_low : na, "OR Low", or_lo_col, linewidth=2, style=plot.style_stepline)

// Margin lines (dashed) — show where breakout actually triggers
plot(or_built and or_valid ? or_high + margin_pts : na, "Breakout Long", color.new(color.lime, 60), linewidth=1, style=plot.style_stepline)
plot(or_built and or_valid ? or_low - margin_pts : na, "Breakout Short", color.new(color.red, 60), linewidth=1, style=plot.style_stepline)

plot(not na(prev_poc) ? prev_poc : na, "Prev POC", color.new(color.fuchsia, 40), linewidth=1, style=plot.style_stepline)
plot(strategy.position_size != 0 and not na(plot_stop) ? plot_stop : na, "Stop", color.orange, linewidth=1, style=plot.style_circles)

bgcolor(in_gather ? color.new(color.blue, 90) : na, title="Gather Period")
bgcolor(or_built and not or_valid ? color.new(color.red, 95) : na, title="Skipped Day (OR out of range)")

// Mode & OR info label
var label info_lbl = na
if barstate.islast
    label.delete(info_lbl)
    string or_info = or_valid ? "VALID" : "SKIPPED"
    float or_rng = nz(or_high) - nz(or_low)
    info_lbl := label.new(bar_index, high,
      "Mode: " + mode +
      "\nOR: " + str.tostring(or_high, "#.##") + " / " + str.tostring(or_low, "#.##") +
      " (" + str.tostring(or_rng, "#.#") + " pts) " + or_info +
      "\nMargin: " + str.tostring(margin_pts, "#.##") + " pts" +
      "\nPrev POC: " + str.tostring(prev_poc, "#.##") +
      "\nAttempts: " + str.tostring(day_attempts) + "/" + str.tostring(max_attempts),
      style=label.style_label_down, color=color.new(color.gray, 70), textcolor=color.white, size=size.small)
