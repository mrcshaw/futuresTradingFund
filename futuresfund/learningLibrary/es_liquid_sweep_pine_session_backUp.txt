//@version=5
// ============================================================
// ES Trend Breakout -- Simple Trend Following
// ============================================================
//
// CONCEPT:
//   SELL: significant low forms → price breaks below it → short
//   BUY:  significant high forms → price breaks above it → long
//   EMA slope confirms the trend direction before entry.
//   Fixed stop + step trailing stop lets runners run.
//
// RULES:
//   1. EMA slope must agree with trade direction
//   2. Pivot high/low identifies the key level
//   3. Candle closes beyond the level = breakout confirmed
//   4. Stop set at fixed distance (configurable)
//   5. Step trailing locks in profit as trade runs
//   6. Session hours filter
// ============================================================

contracts = 1
starting_capital = 25000.00

strategy("ES Trend Breakout safe V2", overlay=true,
  initial_capital=starting_capital, default_qty_type=strategy.fixed, default_qty_value=contracts,
  commission_type=strategy.commission.cash_per_contract, commission_value=4.50,
  slippage=1, pyramiding=0, process_orders_on_close=false, calc_on_every_tick=true)

// ===== EMA SETTINGS =====
ema_len        = input.int(200, "EMA Length")
ema_slope_bars = input.int(10, "EMA Slope Lookback (bars)")
ema_min_angle  = input.float(6.75, "Min EMA Angle (degrees)")

// ===== LEVEL SETTINGS =====
lookback       = input.int(300, "Lookback Window (bars)")

// ===== ENTRY SETTINGS =====
break_buffer   = input.float(0.75, "Break Beyond Level (pts)")

// ===== EXIT SETTINGS =====
pv             = input.float(50.0, "Point Value ($)")
stop_dollars   = input.float(300.0, "Fixed Stop ($)")
use_trail      = input.bool(true, "Enable Trailing Stop")
trail_pct      = input.float(45.0, "Trail: Keep % of Peak Profit")
hard_tp_dollars = input.float(0.0, "Hard TP $ (0 = off)")
// Trail example with 50%:
//   Peak +$200:  stop at +$100 (give back $100)
//   Peak +$600:  stop at +$300 (give back $300)
//   Peak +$1200: stop at +$600 (give back $600)
// Bigger the winner, more room it gets.

// ===== SESSION =====
ny_start_hr    = input.int(9, "NY Session Start Hour (ET)")
ny_start_mn    = input.int(0, "NY Session Start Minute")
ny_end_hr      = input.int(13, "NY Session End Hour (ET)")
ny_end_mn      = input.int(00, "NY Session End Minute")
use_fut_sess   = input.bool(true, "Enable Futures Session")
fut_start_hr   = input.int(20, "Futures Start Hour (ET)")
fut_end_hr     = input.int(08, "Futures End Hour (ET)")
max_trades_day = input.int(3, "Max Trades Per Day")
cool_bars      = input.int(3, "Cooldown After Exit (bars)")
loss_cool_bars = input.int(30, "Cooldown After Loss (bars)")
daily_loss_lim = input.float(500.0, "Daily Loss Limit ($)")

// ===== COMPUTED =====
stop_pts       = stop_dollars / pv
trail_keep     = trail_pct / 100.0
hard_tp_pts    = hard_tp_dollars > 0 ? hard_tp_dollars / pv : 0.0
ema_val        = ta.ema(close, ema_len)

// EMA slope in degrees
float ema_now  = ema_val
float ema_prev = ema_val[ema_slope_bars]
float ema_rise = ema_now - ema_prev
float ema_angle = math.todegrees(math.atan(ema_rise / ema_slope_bars))

// Trend direction from EMA
bool ema_up   = ema_angle >= ema_min_angle and close > ema_val
bool ema_down = ema_angle <= -ema_min_angle and close < ema_val

// ===== SESSION LOGIC =====
bar_hr = hour(time, "America/New_York")
bar_mn = minute(time, "America/New_York")
bar_t  = bar_hr * 100 + bar_mn

ny_start  = ny_start_hr * 100 + ny_start_mn
ny_end    = ny_end_hr * 100 + ny_end_mn
fut_start = fut_start_hr * 100
fut_end   = fut_end_hr * 100

in_ny  = bar_t >= ny_start and bar_t < ny_end
in_fut = use_fut_sess and (bar_t >= fut_start or bar_t < fut_end)
in_session = in_ny or in_fut

prev_hr = hour(time[1], "America/New_York")
prev_mn = minute(time[1], "America/New_York")
prev_t  = prev_hr * 100 + prev_mn
was_ny  = prev_t >= ny_start and prev_t < ny_end
was_fut = use_fut_sess and (prev_t >= fut_start or prev_t < fut_end)
was_in  = was_ny or was_fut
session_ended = was_in and not in_session

// ===== LEVEL TRACKING =====
// After exit: wait `lookback` bars of clean data (no trade bars).
// Then the rolling window only has post-trade price action.
// During a trade: levels freeze. Lines hidden.

var int   bars_since_exit = 999   // starts high so first trade can happen
var float tracked_low     = na
var float tracked_high    = na

// Count bars since last exit
if strategy.position_size[1] != 0 and strategy.position_size == 0
    bars_since_exit := 0
if strategy.position_size == 0
    bars_since_exit += 1

// Only compute levels from clean post-exit bars
// Need `lookback` bars after exit before levels are valid
bool data_clean = bars_since_exit >= lookback

if strategy.position_size == 0 and data_clean
    tracked_low  := ta.lowest(low[1], lookback)
    tracked_high := ta.highest(high[1], lookback)

// ===== DAILY TRACKING =====
var float daily_pnl   = 0.0
var bool  daily_locked = false
var int   last_trades  = 0
var int   prev_day     = -1
var int   trades_today = 0

ny_day = dayofweek(time, "America/New_York")
if ny_day != prev_day and in_session
    prev_day     := ny_day
    daily_pnl    := 0.0
    daily_locked := false
    last_trades  := strategy.closedtrades
    trades_today := 0

if strategy.closedtrades > last_trades
    for i = last_trades to strategy.closedtrades - 1
        daily_pnl += strategy.closedtrades.profit(i)
    last_trades := strategy.closedtrades

if daily_loss_lim > 0 and daily_pnl <= -daily_loss_lim
    daily_locked := true

// ===== FORCE CLOSE =====
if daily_locked and strategy.position_size != 0
    strategy.close_all(comment="Daily Limit")

if session_ended and strategy.position_size != 0
    strategy.close_all(comment="Session Close")

// ===== COOLDOWN =====
var int  last_exit_bar  = -999
var bool last_was_loss  = false

if strategy.position_size[1] != 0 and strategy.position_size == 0
    last_exit_bar := bar_index
    if strategy.closedtrades > 0
        last_was_loss := strategy.closedtrades.profit(strategy.closedtrades - 1) < 0
    else
        last_was_loss := false

int active_cool = last_was_loss ? loss_cool_bars : cool_bars
bool cooled = (bar_index - last_exit_bar) >= active_cool
can_enter = strategy.position_size == 0 and cooled and data_clean and trades_today < max_trades_day and not daily_locked and in_session

// ===== ENTRY -- BREAKOUT OF ROLLING LEVEL =====
// Close makes a new low below the channel → short (EMA confirms)
// Close makes a new high above the channel → long (EMA confirms)
// Uses [1] so we compare to the PREVIOUS bar's channel.
// The current bar's close IS the breakout candle.

var float peak_profit_pts = 0.0
var float current_stop    = 0.0

// SHORT: EMA down + close breaks below tracked low
if can_enter and ema_down and not na(tracked_low)
    if close < tracked_low - break_buffer
        float stop_px = close + stop_pts
        strategy.entry("Short", strategy.short)
        if hard_tp_pts > 0
            strategy.exit("XS", "Short", stop=stop_px, limit=close - hard_tp_pts)
        else
            strategy.exit("XS", "Short", stop=stop_px)
        current_stop    := stop_px
        peak_profit_pts := 0.0
        trades_today   += 1

// LONG: EMA up + close breaks above tracked high
if can_enter and ema_up and not na(tracked_high)
    if close > tracked_high + break_buffer
        float stop_px = close - stop_pts
        strategy.entry("Long", strategy.long)
        if hard_tp_pts > 0
            strategy.exit("XL", "Long", stop=stop_px, limit=close + hard_tp_pts)
        else
            strategy.exit("XL", "Long", stop=stop_px)
        current_stop    := stop_px
        peak_profit_pts := 0.0
        trades_today   += 1

// ===== PERCENTAGE TRAILING STOP =====
// Stop = entry + (peak_profit * trail_keep)
// At 50%: keep half of peak profit, give back half.
// The bigger the winner, the wider the trail -- gives room to run.
if use_trail
    if strategy.position_size > 0
        float avg = strategy.position_avg_price
        float profit_now = high - avg
        if profit_now > peak_profit_pts
            peak_profit_pts := profit_now
        if peak_profit_pts > 0
            float lock_pts = peak_profit_pts * trail_keep
            float new_stop = avg + lock_pts
            if new_stop > current_stop
                current_stop := new_stop
                strategy.exit("XL", "Long", stop=current_stop)

    else if strategy.position_size < 0
        float avg = strategy.position_avg_price
        float profit_now = avg - low
        if profit_now > peak_profit_pts
            peak_profit_pts := profit_now
        if peak_profit_pts > 0
            float lock_pts = peak_profit_pts * trail_keep
            float new_stop = avg - lock_pts
            if new_stop < current_stop
                current_stop := new_stop
                strategy.exit("XS", "Short", stop=current_stop)

// ===== VISUALS =====
// EMA
plot(ema_val, "EMA", color=ema_up ? color.green : ema_down ? color.red : color.gray, linewidth=2)

// Level lines: only shown when flat + data clean + EMA confirms
bool ready_to_show = strategy.position_size == 0 and data_clean
plot(ema_up and ready_to_show ? tracked_high : na, "Break High", color.new(color.aqua, 30), style=plot.style_stepline, linewidth=2)
plot(ema_down and ready_to_show ? tracked_low : na, "Break Low", color.new(color.orange, 30), style=plot.style_stepline, linewidth=2)

// Trailing stop
plot(strategy.position_size != 0 ? current_stop : na, "Trail Stop", color.fuchsia, style=plot.style_stepline, linewidth=2)

// EMA angle in data window
plot(ema_angle, "EMA Angle", color.yellow, display=display.data_window)
plot(trades_today, "Trades Today", color.orange, display=display.data_window)

// Yellow = gathering clean data, blue = ready
bool gathering = strategy.position_size == 0 and not data_clean
bgcolor(gathering ? color.new(color.yellow, 93) : in_session ? color.new(color.blue, 95) : na)
