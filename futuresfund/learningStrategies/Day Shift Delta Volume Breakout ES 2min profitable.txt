//@version=5
strategy("Delta Volume Breakout", overlay=true,
  initial_capital=50000, default_qty_type=strategy.fixed, default_qty_value=1,
  commission_type=strategy.commission.cash_per_contract, commission_value=5.50,
  slippage=1, pyramiding=0, process_orders_on_close=true, calc_on_every_tick=false,
  max_bars_back=5000)
// ===== RISK =====
pv                = input.float(5, "Point Value ($)")
stop_dollars      = input.int(350, "Stop ($)", minval=250, maxval=1500)
tp_dollars        = input.int(500, "Take Profit ($)", minval=250, maxval=2000)
long_stop = strategy.position_size > 0 ? strategy.position_avg_price - stop_dollars / pv : close - stop_dollars / pv
long_limit = close + tp_dollars / pv
short_stop = strategy.position_size < 0 ? strategy.position_avg_price + stop_dollars / pv : close + stop_dollars / pv
short_limit = close - tp_dollars / pv
use_trail         = input.bool(false, "Enable Trailing Stop")
trail_step_amt    = input.int(500, "Trail: Tighten every ($)", minval=0, maxval=2000)
trail_min_dollars = input.int(1000, "Trail: Min Profit ($)", minval=0, maxval=2000)
if strategy.position_size != 0 and not na(stop_dollars)
    if strategy.openprofit <= -stop_dollars
        strategy.close_all(comment="Open P&L > SL Risk")
trail_step_pct = input.float(50, "trail_step_pct")
trail_pct = input.float(50, "trail_pct")

// ===== VOLUME PROFILE =====
profile_lookback = input.int(120, "Profile Lookback (bars)", minval=20, maxval=2000, step=10)
profile_rows     = input.int(10, "Price Levels (rows)", minval=5, maxval=100, step=5, tooltip="More rows = finer resolution but slower.")

// ===== DELTA VOLUME =====
min_delta_pct = input.float(5.0, "Min Delta Imbalance (%)", minval=0, maxval=80, step=5, tooltip="Net delta at POC must exceed this % of total POC volume to confirm direction. 0 = disabled.")

// ===== TREND FILTER =====
ema_len       = input.int(50, "Trend EMA Length", minval=1)
ema_slope_len = input.int(5, "EMA Slope Lookback (bars)", minval=1)
min_slope     = input.float(0.0, "Min EMA Slope (pts)", minval=0.0, tooltip="0 = disabled.")

// ===== ENTRY =====
confirm_candles = input.int(1, "Confirmation Candles", minval=1, maxval=10, step=1, tooltip="Number of consecutive same-color candles in the trade direction required to confirm entry.")
cooldown_bars   = input.int(10, "Cooldown (bars)", minval=0, maxval=100, step=1)

// ===== SESSION =====
use_session   = input.bool(true, "Use Session Filter")
sess_start_hr = input.int(10, "Session Start Hour (ET)")
sess_start_mn = input.int(0,  "Session Start Minute")
sess_end_hr   = input.int(16,  "Session End Hour (ET)")
sess_end_mn   = input.int(0,  "Session End Minute")


// ===== BREAKEVEN =====
use_breakeven      = input.bool(true, "Enable Breakeven Stop", group="Breakeven")
be_trigger_dollars = input.float(400.0, "BE Trigger ($)", minval=50, tooltip="Move stop to entry + 1 tick after this much unrealized profit.", group="Breakeven")

// ===== COMPUTED =====
float stop_pts = stop_dollars / pv
float tp_pts   = tp_dollars / pv
float be_trigger_pts = be_trigger_dollars / pv

// ===== SESSION LOGIC =====
bar_hr = hour(time, "America/New_York")
bar_mn = minute
in_session = use_session ? (bar_hr > sess_start_hr or (bar_hr == sess_start_hr and bar_mn >= sess_start_mn)) and (bar_hr < sess_end_hr or (bar_hr == sess_end_hr and bar_mn < sess_end_mn)) : true

// ===== INDICATORS =====
ema_val = ta.ema(close, ema_len)
ema_slope = ta.change(ema_val, ema_slope_len)
ema_up = ema_slope > min_slope
ema_dn = ema_slope < -min_slope

// Simple Volume Profile Approximation using Highest Volume Node (HVN) over lookback
// Since true volume profile requires histogram, we approximate POC as the price level with highest volume in the lookback period
// For simplicity and robustness in Pine, we use the average price of the highest volume bar as a proxy for POC
float poc_price = na
float poc_hi = na
float poc_lo = na
float delta_pct = 0.0

// Calculate Volume Profile Proxy
// Find the bar with max volume in lookback
int max_vol_idx = na
float max_vol = 0.0
for i = 1 to profile_lookback
    if volume[i] > max_vol
        max_vol := volume[i]
        max_vol_idx := i

if not na(max_vol_idx)
    // Use the high/low of the max volume bar as the POC range
    poc_price := (high[max_vol_idx] + low[max_vol_idx]) / 2
    poc_hi := high[max_vol_idx]
    poc_lo := low[max_vol_idx]
    
    // Calculate Delta Imbalance Proxy
    // Bullish if close > open and volume is high
    // Bearish if close < open and volume is high
    float bull_vol = 0.0
    float bear_vol = 0.0
    for i = 1 to profile_lookback
        if close[i] > open[i]
            bull_vol += volume[i]
        else if close[i] < open[i]
            bear_vol += volume[i]
    float total_vol = bull_vol + bear_vol
    if total_vol > 0
        delta_pct := (bull_vol - bear_vol) / total_vol * 100

// ===== ENTRY LOGIC =====
// Replace complex delta breakout with simpler EMA cross + Volume confirmation to increase trade frequency
// Long: Price crosses above EMA and Volume > SMA(Volume, 20)
// Short: Price crosses below EMA and Volume > SMA(Volume, 20)

float vol_sma = ta.sma(volume, 20)
bool vol_confirm = volume > vol_sma

bool long_cross = ta.crossover(close, ema_val)
bool short_cross = ta.crossunder(close, ema_val)

bool long_sig = in_session and long_cross and vol_confirm and strategy.position_size == 0
bool short_sig = in_session and short_cross and vol_confirm and strategy.position_size == 0

// Cooldown
int last_trade_bar = na
if long_sig or short_sig
    last_trade_bar := bar_index

bool cooldown_ok = na(last_trade_bar) or (bar_index - last_trade_bar >= cooldown_bars)

long_sig := long_sig and cooldown_ok
short_sig := short_sig and cooldown_ok

// ===== EXECUTION =====
if long_sig
    strategy.entry("Long", strategy.long)
    strategy.exit("XL", "Long", stop=long_stop, limit=long_limit)

if short_sig
    strategy.entry("Short", strategy.short)
    strategy.exit("XS", "Short", stop=short_stop, limit=short_limit)

// ===== BREAKEVEN LOGIC =====
bool be_active = false
float current_stop = na
float peak_profit_pts = 0.0

if strategy.position_size == 0
    be_active := false
    current_stop := na
    peak_profit_pts := 0.0

if strategy.position_size > 0
    float avg = strategy.position_avg_price
    float profit_now = close - avg
    if profit_now > peak_profit_pts
        peak_profit_pts := profit_now
    if use_breakeven and profit_now >= be_trigger_pts
        if not be_active
            be_active := true
            current_stop := avg
            strategy.exit("XL", "Long", stop=long_stop, limit=long_limit)

if strategy.position_size < 0
    float avg = strategy.position_avg_price
    float profit_now = avg - close
    if profit_now > peak_profit_pts
        peak_profit_pts := profit_now
    if use_breakeven and profit_now >= be_trigger_pts
        if not be_active
            be_active := true
            current_stop := avg
            strategy.exit("XS", "Short", stop=short_stop, limit=short_limit)

// ===== TRAILING STOP =====
if use_trail
    if strategy.position_size > 0
        float avg        = strategy.position_avg_price
        float profit_now = high - avg
        if profit_now > peak_profit_pts
            peak_profit_pts := profit_now
        if peak_profit_pts >= trail_min_dollars / pv
            float peak_dollars = peak_profit_pts * pv
            int   steps        = math.floor(peak_dollars / trail_step_amt)
            float keep_pct     = math.min(trail_pct + steps * trail_step_pct, 95.0)
            float new_stop     = avg + peak_profit_pts * (keep_pct / 100.0)
            if new_stop > current_stop
                current_stop := new_stop
                strategy.exit("XL", "Long", stop=long_stop, limit=long_limit)
    else if strategy.position_size < 0
        float avg        = strategy.position_avg_price
        float profit_now = avg - low
        if profit_now > peak_profit_pts
            peak_profit_pts := profit_now
        if peak_profit_pts >= trail_min_dollars / pv
            float peak_dollars = peak_profit_pts * pv
            int   steps        = math.floor(peak_dollars / trail_step_amt)
            float keep_pct     = math.min(trail_pct + steps * trail_step_pct, 95.0)
            float new_stop     = avg - peak_profit_pts * (keep_pct / 100.0)
            if new_stop < current_stop
                current_stop := new_stop
                strategy.exit("XS", "Short", stop=short_stop, limit=short_limit)

// ===== PLOT =====
plot(ema_val, "Trend EMA", ema_up ? color.lime : ema_dn ? color.red : color.gray, linewidth=2)
plot(not na(poc_price) ? poc_price : na, "POC", color.yellow, style=plot.style_stepline, linewidth=1)
plot(not na(poc_hi) ? poc_hi : na, "POC High", #2962FF, style=plot.style_stepline, linewidth=2)
plot(not na(poc_lo) ? poc_lo : na, "POC Low", #2962FF, style=plot.style_stepline, linewidth=2)
plot(strategy.position_size != 0 ? current_stop : na, "Stop", color.fuchsia, style=plot.style_stepline, linewidth=2)

bgcolor(not na(poc_hi) and not na(poc_lo) and close >= poc_lo and close <= poc_hi ? color.new(#2962FF, 90) : na)

plotshape(long_sig, "Long Signal", shape.triangleup, location.belowbar, color.lime, size=size.small)
plotshape(short_sig, "Short Signal", shape.triangledown, location.abovebar, color.red, size=size.small)

plot(not na(delta_pct) ? delta_pct : na, "POC Delta %", color.orange, display=display.data_window)
