//@version=5
// ICT Unicorn + ORB Hybrid — ORB + FVG confirmation
// Consistent: require FVG in breakout direction | Profit: FVG optional
strategy("ICT Unicorn + ORB 100k", overlay=true,
  initial_capital=50000, default_qty_type=strategy.fixed, default_qty_value=1,
  commission_type=strategy.commission.cash_per_contract, commission_value=5.50,
  slippage=1, pyramiding=0, process_orders_on_close=false, calc_on_every_tick=true,
  max_bars_back=5000)
// ===== RISK =====
pv                = input.float(100, "Point Value ($)")
stop_dollars      = input.int(250, "Stop ($)", minval=250, maxval=750)
tp_dollars        = input.int(500, "Take Profit ($)", minval=250)
long_stop = strategy.position_size > 0 ? strategy.position_avg_price - stop_dollars / pv : close - stop_dollars / pv
long_limit = close + tp_dollars / pv
short_stop = strategy.position_size < 0 ? strategy.position_avg_price + stop_dollars / pv : close + stop_dollars / pv
short_limit = close - tp_dollars / pv
if strategy.position_size != 0 and not na(stop_dollars)
    if strategy.openprofit <= -stop_dollars
        strategy.close_all(comment="Open P&L > SL Risk")

// ===== SESSION =====
use_session   = input.bool(true, "Use Session Filter")
sess_start_hr = input.int(18, "Session Start Hour (ET)")
sess_start_mn = input.int(0,  "Session Start Minute")
sess_end_hr   = input.int(16,  "Session End Hour (ET)")
sess_end_mn   = input.int(0,  "Session End Minute")

// ===== SKIP WINDOW (a time of day that keeps losing) =====
use_skip      = input.bool(true, "Skip This Time Of Day", group="Skip Window", tooltip="No trade opens inside this window, and an open trade is closed when it starts.")
skip_start_hr = input.int(10, "Skip Start Hour (ET)", minval=0, maxval=23, group="Skip Window")
skip_start_mn = input.int(0, "Skip Start Minute", minval=0, maxval=59, group="Skip Window")
skip_end_hr   = input.int(11, "Skip End Hour (ET)", minval=0, maxval=23, group="Skip Window")
skip_end_mn   = input.int(0, "Skip End Minute", minval=0, maxval=59, group="Skip Window")
skip_bar_t = hour(time, "America/New_York") * 100 + minute(time, "America/New_York")
skip_from = skip_start_hr * 100 + skip_start_mn
skip_to = skip_end_hr * 100 + skip_end_mn
in_skip_window = use_skip and (skip_from > skip_to ? (skip_bar_t >= skip_from or skip_bar_t < skip_to) : (skip_bar_t >= skip_from and skip_bar_t < skip_to))
skip_window_open = not in_skip_window
if in_skip_window and strategy.position_size != 0
    strategy.close_all(comment="Skip Window")

sess_bar_hr = hour(time, "America/New_York")
sess_bar_mn = minute(time, "America/New_York")
sess_bar_t = sess_bar_hr * 100 + sess_bar_mn
sess_clock_start = sess_start_hr * 100 + sess_start_mn
sess_clock_end = sess_end_hr * 100 + sess_end_mn
in_session = not use_session or (sess_clock_start > sess_clock_end ? (sess_bar_t >= sess_clock_start or sess_bar_t < sess_clock_end) : (sess_bar_t >= sess_clock_start and sess_bar_t < sess_clock_end))
if not in_session
    strategy.close_all(comment="Session")

// ===== MODE =====
mode = input.string("Profit", "Mode", options=["Profit","Consistent"],
  tooltip="Profit: FVG optional, more trades.\nConsistent: FVG required, fewer trades.")

// ===== TIMING =====
gather_min   = input.int(15, "Gather Minutes", minval=5, maxval=60, step=5)
entry_end_hr = input.int(11, "Entry End Hour", minval=10, maxval=14)
entry_end_mn = input.int(0, "Entry End Minute", minval=0, maxval=59, step=5)
eod_hr       = input.int(16, "EOD Flatten Hour")
eod_mn       = input.int(30, "EOD Flatten Min")

// ===== FILTERS =====
i_margin     = input.float(1.0, "Breakout Margin (pts)", minval=0, step=0.25)
i_or_min     = input.float(0.0, "OR Min Size (pts)", minval=0, step=1)
i_or_max     = input.float(9999.0, "OR Max Size (pts)", minval=1, step=5)
max_attempts = input.int(2, "Max Attempts / Day", minval=1, maxval=5)

// ===== FVG (ICT Unicorn) =====
fvg_lookback = input.int(30, "FVG Lookback (bars)", minval=10, maxval=100)
fvg_min_atr  = input.float(0.04, "FVG Min Size (ATR ratio)", minval=0.02, step=0.01)
atr_len      = input.int(14, "ATR Length")

// ===== TIME HELPERS =====
is_new_day = ta.change(time("D")) != 0
t = time("1", "0930-1600", "America/New_York")
in_rth = not na(t)

hrs = hour(time, "America/New_York")
mns = minute(time, "America/New_York")
float mins_from_open = (hrs - 9) * 60.0 + (mns - 30)
bool in_gather = in_rth and mins_from_open >= 0 and mins_from_open < gather_min
bool gather_done = in_rth and mins_from_open >= gather_min
int bar_time = hrs * 100 + mns
bool in_entry_window = gather_done and bar_time < (entry_end_hr * 100 + entry_end_mn)
bool at_eod = bar_time >= (eod_hr * 100 + eod_mn)

// ===== ATR =====
atr = ta.atr(atr_len)

// ===== FVG DETECTION =====
// Bullish FVG: low[2] > high (gap up)
// Bearish FVG: high[2] < low (gap down)
bool bull_fvg = low[2] > high and (low[2] - high) >= atr * fvg_min_atr
bool bear_fvg = high[2] < low and (low - high[2]) >= atr * fvg_min_atr

// Recent FVG: rolling counter when FVG detected
var int recent_bull_cnt = 0
var int recent_bear_cnt = 0
if bull_fvg
    recent_bull_cnt := fvg_lookback
else if recent_bull_cnt > 0
    recent_bull_cnt := recent_bull_cnt - 1
if bear_fvg
    recent_bear_cnt := fvg_lookback
else if recent_bear_cnt > 0
    recent_bear_cnt := recent_bear_cnt - 1

bool recent_bull_fvg = recent_bull_cnt > 0
bool recent_bear_fvg = recent_bear_cnt > 0

// ===== OPENING RANGE =====
var float or_high = na
var float or_low  = na
var bool  or_built = false
var bool  or_valid = false

if is_new_day
    or_high  := na
    or_low   := na
    or_built := false
    or_valid := false

if in_gather
    or_high := na(or_high) ? high : math.max(or_high, high)
    or_low  := na(or_low)  ? low  : math.min(or_low, low)

if gather_done and not or_built
    or_built := true
    float or_range = nz(or_high) - nz(or_low)
    or_valid := or_range >= i_or_min and or_range <= i_or_max and or_range > 0

// ===== ATTEMPT COUNTER =====
var int day_attempts = 0
if is_new_day
    day_attempts := 0

// ===== PLOT STOP STATE =====
var float plot_stop = na

if is_new_day
    plot_stop := na

// ===== EOD FLATTEN =====
if at_eod and strategy.position_size != 0
    strategy.close_all(comment="EOD")

// ===== ENTRY SIGNALS =====
bool can_trade = in_entry_window and or_built and or_valid and day_attempts < max_attempts

bool breakout_long  = can_trade and close > (or_high + i_margin)
bool breakout_short = can_trade and close < (or_low - i_margin)

// FVG filter: Consistent mode requires FVG in breakout direction
if mode == "Consistent"
    if not recent_bull_fvg
        breakout_long := false
    if not recent_bear_fvg
        breakout_short := false

bool long_sig  = breakout_long and strategy.position_size == 0
bool short_sig = breakout_short and strategy.position_size == 0

// ===== ENTRY + EXIT =====
bool entry_ok = barstate.isconfirmed

if long_sig and entry_ok
    day_attempts += 1
    if in_session
        if skip_window_open
            strategy.entry("Long", strategy.long)
        strategy.exit("Long Exit", "Long", stop=long_stop, limit=long_limit)
        float long_stop = close - stop_dollars / pv
        float long_limit = close + tp_dollars / pv
        plot_stop := long_stop
        strategy.exit("Long Exit", "Long", stop=long_stop, limit=long_limit, comment_loss="SL", comment_profit="TP")
else if short_sig and entry_ok
    day_attempts += 1
    if in_session
        if skip_window_open
            strategy.entry("Short", strategy.short)
        strategy.exit("Short Exit", "Short", stop=short_stop, limit=short_limit)
        float short_stop = close + stop_dollars / pv
        float short_limit = close - tp_dollars / pv
        plot_stop := short_stop
        strategy.exit("Short Exit", "Short", stop=short_stop, limit=short_limit, comment_loss="SL", comment_profit="TP")
else if strategy.position_size > 0
    float avg = na(strategy.position_avg_price) ? close : strategy.position_avg_price
    float long_stop = avg - stop_dollars / pv
    float long_limit = close + tp_dollars / pv
    plot_stop := long_stop
    strategy.exit("Long Exit", "Long", stop=long_stop, limit=long_limit, comment_loss="SL", comment_profit="TP")
else if strategy.position_size < 0
    float avg = na(strategy.position_avg_price) ? close : strategy.position_avg_price
    float short_stop = avg + stop_dollars / pv
    float short_limit = close - tp_dollars / pv
    plot_stop := short_stop
    strategy.exit("Short Exit", "Short", stop=short_stop, limit=short_limit, comment_loss="SL", comment_profit="TP")

// ===== RESET ON FLAT =====
if strategy.position_size == 0 and strategy.position_size[1] != 0
    plot_stop := na

// ===== PLOTS =====
color or_hi_col = or_built and or_valid ? color.new(color.lime, 20) : or_built ? color.new(color.gray, 60) : na
color or_lo_col = or_built and or_valid ? color.new(color.red, 20) : or_built ? color.new(color.gray, 60) : na

plot(or_built ? or_high : na, "OR High", or_hi_col, linewidth=2, style=plot.style_stepline)
plot(or_built ? or_low : na, "OR Low", or_lo_col, linewidth=2, style=plot.style_stepline)
plot(or_built and or_valid ? or_high + i_margin : na, "Breakout Long", color.new(color.lime, 60), linewidth=1, style=plot.style_stepline)
plot(or_built and or_valid ? or_low - i_margin : na, "Breakout Short", color.new(color.red, 60), linewidth=1, style=plot.style_stepline)
plot(strategy.position_size != 0 and not na(plot_stop) ? plot_stop : na, "Stop", color.orange, linewidth=1, style=plot.style_circles)

bgcolor(in_gather ? color.new(color.blue, 90) : na, title="Gather Period")
bgcolor(or_built and not or_valid ? color.new(color.red, 95) : na, title="Skipped Day")

// FVG highlights
bgcolor(bull_fvg ? color.new(color.lime, 85) : na, title="Bull FVG")
bgcolor(bear_fvg ? color.new(color.red, 85) : na, title="Bear FVG")

var label info_lbl = na
if barstate.islast
    label.delete(info_lbl)
    string or_info = or_valid ? "VALID" : "SKIPPED"
    float or_rng = nz(or_high) - nz(or_low)
    info_lbl := label.new(bar_index, high,
      "Mode: " + mode + " | " + or_info +
      "\nOR: " + str.tostring(or_high, "#.##") + " / " + str.tostring(or_low, "#.##") +
      " (" + str.tostring(or_rng, "#.#") + " pts)" +
      "\nFVG Bull: " + str.tostring(recent_bull_fvg) + " Bear: " + str.tostring(recent_bear_fvg) +
      "\nAttempts: " + str.tostring(day_attempts) + "/" + str.tostring(max_attempts),
      style=label.style_label_down, color=color.new(color.gray, 70), textcolor=color.white, size=size.small)
