//@version=5
strategy("Smart Trail Signals", overlay=true,
  initial_capital=50000, default_qty_type=strategy.fixed, default_qty_value=1,
  commission_type=strategy.commission.cash_per_contract, commission_value=5.50,
  slippage=1, pyramiding=0, process_orders_on_close=true, calc_on_every_tick=false)
// ===== RISK =====
pv                = input.float(5, "Point Value ($)")
stop_dollars      = input.int(250, "Stop ($)", minval=250, maxval=1500)
tp_dollars        = input.int(500, "Take Profit ($)", minval=250, maxval=2000)
use_trail         = input.bool(false, "Enable Trailing Stop")
trail_step_amt    = input.int(500, "Trail: Tighten every ($)", minval=0, maxval=2000)
trail_min_dollars = input.int(1000, "Trail: Min Profit ($)", minval=0, maxval=2000)
trail_pct         = input.float(50, "Trail: Keep % of Peak Profit")
trail_step_pct    = input.float(50, "Trail: Tighten by X% per step")
long_stop = strategy.position_size > 0 ? strategy.position_avg_price - stop_dollars / pv : close - stop_dollars / pv
long_limit = close + tp_dollars / pv
short_stop = strategy.position_size < 0 ? strategy.position_avg_price + stop_dollars / pv : close + stop_dollars / pv
short_limit = close - tp_dollars / pv
if strategy.position_size != 0 and not na(stop_dollars)
    if strategy.openprofit <= -stop_dollars
        strategy.close_all(comment="Open P&L > SL Risk")

// ===== SIGNAL SETTINGS =====
sensitivity = input.int(13, "Optimal Sensitivity", minval=1, tooltip="Adjusts the agility of the Smart Trail.")
min_score   = input.int(2, "Min AI Score to Trade (1-4)", minval=1, maxval=4, tooltip="Only take signals with this score or higher.")

// ===== TREND FILTER =====
ema_len       = input.int(400, "Trend EMA Length", minval=1)
ema_slope_len = input.int(5, "EMA Slope Lookback (bars)", minval=1)
min_slope     = input.float(0.2, "Min EMA Slope (pts)", minval=0.0, tooltip="Minimum slope to confirm trending. 0 = disabled.")
pullback_dollars = input.float(125.0, "Pullback Zone ($)", tooltip="How close price must be to 200 EMA for pullback entry.")

// ===== SESSION =====
use_session   = input.bool(true, "Use Session Filter")
sess_start_hr = input.int(18, "Session Start Hour (ET)")
sess_start_mn = input.int(0,  "Session Start Minute")
sess_end_hr   = input.int(8,  "Session End Hour (ET)")
sess_end_mn   = input.int(0,  "Session End Minute")

// ===== COMPUTED =====
float stop_pts     = stop_dollars / pv
float tp_pts       = tp_dollars / pv
float pullback_pts = pullback_dollars / pv

// ===== SESSION LOGIC =====
bar_hr = hour(time, "America/New_York")
bar_mn = minute(time, "America/New_York")
bar_t  = bar_hr * 100 + bar_mn
int s_start = sess_start_hr * 100 + sess_start_mn
int s_end   = sess_end_hr * 100 + sess_end_mn

bool in_session = not use_session or (s_start > s_end ? (bar_t >= s_start or bar_t < s_end) : (bar_t >= s_start and bar_t < s_end))

prev_hr = hour(time[1], "America/New_York")
prev_mn = minute(time[1], "America/New_York")
prev_t  = prev_hr * 100 + prev_mn
bool was_in_session = not use_session or (s_start > s_end ? (prev_t >= s_start or prev_t < s_end) : (prev_t >= s_start and prev_t < s_end))
bool session_ended = was_in_session and not in_session

// ===== INDICATORS =====
ema_val = ta.ema(close, ema_len)
ema_slope = (ema_val - ema_val[ema_slope_len]) / ema_slope_len
bool ema_up = ema_slope > min_slope
bool ema_dn = ema_slope < -min_slope

// Simple RSI for entry trigger
rsi_len = 14
rsi_val = ta.rsi(close, rsi_len)

// Entry Logic: Replace complex AI score with RSI extremes aligned with trend
// Long: Trend Up (EMA slope > min_slope) AND RSI < 30 (Oversold)
// Short: Trend Down (EMA slope < -min_slope) AND RSI > 70 (Overbought)
bool long_ok = in_session and ema_up and rsi_val < 30 and strategy.position_size == 0
bool short_ok = in_session and ema_dn and rsi_val > 70 and strategy.position_size == 0

// Flatten at 16:45 New York
bool flatten_time = bar_hr == 16 and bar_mn >= 45
if flatten_time and strategy.position_size != 0
    strategy.close_all(comment="EOD Flatten")

// ===== EXECUTION =====
var float current_stop = na
var float peak_profit_pts = 0.0
var bool armed_long = false
var bool armed_short = false

if long_ok
    entry_price    := close
    current_stop   := close - stop_pts
    peak_profit_pts := 0.0
    armed_long     := true
    strategy.entry("Long", strategy.long)
    strategy.exit("XL", "Long", stop=long_stop, limit=long_limit)
else if short_ok
    entry_price    := close
    current_stop   := close + stop_pts
    peak_profit_pts := 0.0
    armed_short    := true
    strategy.entry("Short", strategy.short)
    strategy.exit("XS", "Short", stop=short_stop, limit=short_limit)

// ===== TRAILING STOP (only when enabled; keeps TP in place) =====
if use_trail and strategy.position_size != 0
    float avg = strategy.position_avg_price
    if strategy.position_size > 0
        float profit_now = high - avg
        if profit_now > peak_profit_pts
            peak_profit_pts := profit_now
        if peak_profit_pts >= trail_min_dollars / pv
            float peak_dollars = peak_profit_pts * pv
            int   steps        = math.floor(peak_dollars / trail_step_amt)
            float keep_pct     = math.min(trail_pct + steps * trail_step_pct, 95.0)
            float new_stop     = avg + peak_profit_pts * (keep_pct / 100.0)
            if new_stop > current_stop
                current_stop := new_stop
                strategy.exit("XL", "Long", stop=long_stop, limit=long_limit)
    else
        float profit_now = avg - low
        if profit_now > peak_profit_pts
            peak_profit_pts := profit_now
        if peak_profit_pts >= trail_min_dollars / pv
            float peak_dollars = peak_profit_pts * pv
            int   steps        = math.floor(peak_dollars / trail_step_amt)
            float keep_pct     = math.min(trail_pct + steps * trail_step_pct, 95.0)
            float new_stop     = avg - peak_profit_pts * (keep_pct / 100.0)
            if new_stop < current_stop
                current_stop := new_stop
                strategy.exit("XS", "Short", stop=short_stop, limit=short_limit)

// ===== PLOT =====
plot(ema_val, "Trend EMA", ema_up ? color.lime : ema_dn ? color.red : color.gray, linewidth=2)
plotshape(long_ok, "Buy Signal", shape.triangleup, location.belowbar, #00e676, size=size.small)
plotshape(short_ok, "Sell Signal", shape.triangledown, location.abovebar, #ff5252, size=size.small)
