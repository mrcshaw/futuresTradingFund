//@version=2
strategy("Stochastic + RSI, Double Strategy (by ChartArt)", shorttitle="CA_-_RSI_Stoch_Strat", overlay=true)
// ===== RISK =====
pv                = input.float(50, "Point Value ($)")
stop_dollars      = input.int(250, "Stop ($)", minval=50, maxval=700)
tp_dollars        = input.int(500, "Take Profit ($)", minval=50)
use_trail         = input.bool(false, "Enable Trailing Stop")
trail_pct         = input.float(50, "Trail keep (%)", minval=0, maxval=100)
trail_step_amt    = input.int(500, "Trail: Tighten every ($)", minval=0, maxval=2000)
trail_step_pct    = input.float(50, "Trail step (%)", minval=0, maxval=100)
trail_min_dollars = input.int(1000, "Trail: Min Profit ($)", minval=0, maxval=2000)
long_stop = strategy.position_size > 0 ? strategy.position_avg_price - stop_dollars / pv : close - stop_dollars / pv
long_limit = close + tp_dollars / pv
short_stop = strategy.position_size < 0 ? strategy.position_avg_price + stop_dollars / pv : close + stop_dollars / pv
short_limit = close - tp_dollars / pv
if strategy.position_size != 0 and not na(stop_dollars)
    if strategy.openprofit <= -stop_dollars
        strategy.close_all(comment="Open P&L > SL Risk")
// ===== SESSION =====
use_session   = input.bool(true, "Use Session Filter")
sess_start_hr = input.int(18, "Session Start Hour (ET)")
sess_start_mn = input.int(0,  "Session Start Minute")
sess_end_hr   = input.int(16,  "Session End Hour (ET)")
sess_end_mn   = input.int(0,  "Session End Minute")
sess_bar_hr = hour(time, "America/New_York")
sess_bar_mn = minute(time, "America/New_York")
sess_bar_t = sess_bar_hr * 100 + sess_bar_mn
sess_clock_start = sess_start_hr * 100 + sess_start_mn
sess_clock_end = sess_end_hr * 100 + sess_end_mn
in_session = not use_session or (sess_clock_start > sess_clock_end ? (sess_bar_t >= sess_clock_start or sess_bar_t < sess_clock_end) : (sess_bar_t >= sess_clock_start and sess_bar_t < sess_clock_end))
if not in_session
    strategy.close_all(comment="Session")

// ChartArt's Stochastic Slow + Relative Strength Index, Double Strategy
//
// Version 1.0
// Idea by ChartArt on October 23, 2015.
//
// This strategy combines the classic RSI
// strategy to sell when the RSI increases
// over 70 (or to buy when it falls below 30),
// with the classic Stochastic Slow strategy
// to sell when the Stochastic oscillator
// exceeds the value of 80 (and to buy when
// this value is below 20).
//
// This simple strategy only triggers when
// both the RSI and the Stochastic are together
// in overbought or oversold conditions.
//
// List of my work: 
// https://www.tradingview.com/u/ChartArt/


///////////// Stochastic Slow
Stochlength = input(14, minval=1, title="lookback length of Stochastic")
StochOverBought = input(80, title="Stochastic overbought condition")
StochOverSold = input(20, title="Stochastic oversold condition")
smoothK = input(3, title="smoothing of Stochastic %K ")
smoothD = input(3, title="moving average of Stochastic %K")
k = sma(stoch(close, high, low, Stochlength), smoothK)
d = sma(k, smoothD)

 
///////////// RSI 
RSIlength = input( 14, minval=1 , title="lookback length of RSI")
RSIOverBought = input( 70  , title="RSI overbought condition")
RSIOverSold = input( 30  , title="RSI oversold condition")
RSIprice = close
vrsi = rsi(RSIprice, RSIlength)


///////////// Double strategy: RSI strategy + Stochastic strategy

if (not na(k) and not na(d))
    if (crossover(k,d) and k < StochOverSold)
        if (not na(vrsi)) and (crossover(vrsi, RSIOverSold))
            if in_session
                strategy.entry("LONG", strategy.long, comment="StochLE + RsiLE")
 
 
if (crossunder(k,d) and k > StochOverBought)
    if (crossunder(vrsi, RSIOverBought))
        if in_session
            strategy.entry("SHORT", strategy.short, comment="StochSE + RsiSE")
