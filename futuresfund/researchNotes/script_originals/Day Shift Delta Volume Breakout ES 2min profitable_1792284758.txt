//@version=5
strategy("Delta Volume Breakout", overlay=true,
  initial_capital=50000, default_qty_type=strategy.fixed, default_qty_value=1,
  commission_type=strategy.commission.cash_per_contract, commission_value=5.50,
  slippage=1, pyramiding=0, process_orders_on_close=true, calc_on_every_tick=false,
  max_bars_back=5000)
// ===== RISK =====
pv                = input.float(50, "Point Value ($)")
stop_dollars      = input.int(250, "Stop ($)", minval=250, maxval=1500)
tp_dollars        = input.int(500, "Take Profit ($)", minval=250, maxval=2000)
long_stop = close - stop_dollars / pv
long_limit = close + tp_dollars / pv
short_stop = close + stop_dollars / pv
short_limit = close - tp_dollars / pv
use_trail         = input.bool(false, "Enable Trailing Stop")
trail_step_amt    = input.int(500, "Trail: Tighten every ($)", minval=0, maxval=2000)
trail_min_dollars = input.int(1000, "Trail: Min Profit ($)", minval=0, maxval=2000)
if strategy.position_size != 0 and not na(stop_dollars)
    if strategy.openprofit <= -stop_dollars
        strategy.close_all(comment="Open P&L > SL Risk")
trail_step_pct = input.float(50, "trail_step_pct")
trail_pct = input.float(50, "trail_pct")

// ===== VOLUME PROFILE =====
profile_lookback = input.int(120, "Profile Lookback (bars)", minval=20, maxval=2000, step=10)
profile_rows     = input.int(10, "Price Levels (rows)", minval=5, maxval=100, step=5, tooltip="More rows = finer resolution but slower.")

// ===== DELTA VOLUME =====
min_delta_pct = input.float(5, "Min Delta Imbalance (%)", minval=0, maxval=80, step=5, tooltip="Net delta at POC must exceed this % of total POC volume to confirm direction. 0 = disabled.")

// ===== TREND FILTER =====
ema_len       = input.int(50, "Trend EMA Length", minval=1)
ema_slope_len = input.int(5, "EMA Slope Lookback (bars)", minval=1)
min_slope     = input.float(0, "Min EMA Slope (pts)", minval=0.0, tooltip="0 = disabled.")

// ===== ENTRY =====
// confirm_candles removed from logic, kept as input for compatibility but unused in new entry
confirm_candles = input.int(3, "Confirmation Candles", minval=1, maxval=10, step=1, tooltip="Ignored in new entry logic.")
cooldown_bars   = input.int(10, "Cooldown (bars)", minval=0, maxval=100, step=1)

// ===== SESSION =====
use_session   = input.bool(true, "Use Session Filter")
sess_start_hr = input.int(8, "Session Start Hour (ET)")
sess_start_mn = input.int(0,  "Session Start Minute")
sess_end_hr   = input.int(16,  "Session End Hour (ET)")
sess_end_mn   = input.int(0,  "Session End Minute")


// ===== BREAKEVEN =====
use_breakeven      = input.bool(true, "Enable Breakeven Stop", group="Breakeven")
be_trigger_dollars = input.float(200, "BE Trigger ($)", minval=50, tooltip="Move stop to entry + 1 tick after this much unrealized profit.", group="Breakeven")

// ===== COMPUTED =====
float stop_pts = stop_dollars / pv
float tp_pts   = tp_dollars / pv
float be_trigger_pts = be_trigger_dollars / pv

// ===== SESSION LOGIC =====
bar_hr = hour(time, "America/New_York")
bar_mn = minute(time, "America/New_York")
in_session = use_session ? (bar_hr > sess_start_hr or (bar_hr == sess_start_hr and bar_mn >= sess_start_mn)) and (bar_hr < sess_end_hr or (bar_hr == sess_end_hr and bar_mn < sess_end_mn)) : true

// ===== VOLUME PROFILE CALCULATIONS =====
// Simplified POC calculation for performance and clarity
float poc_price = na
float poc_hi = na
float poc_lo = na
float poc_delta_pct = na

if profile_lookback > 0
    float high_range = high[profile_lookback-1]
    float low_range = low[profile_lookback-1]
    for i = 1 to profile_lookback - 1
        high_range := math.max(high_range, high[i])
        low_range := math.min(low_range, low[i])
    
    if high_range > low_range
        float range_size = high_range - low_range
        float row_size = range_size / profile_rows
        float poc_vol = 0
        float max_vol = 0
        float current_poc = na
        
        for r = 0 to profile_rows - 1
            float row_low = low_range + r * row_size
            float row_high = row_low + row_size
            
            float vol_in_row = 0
            for j = 0 to profile_lookback - 1
                if close[j] >= row_low and close[j] < row_high
                    vol_in_row += volume[j]
            
            if vol_in_row > max_vol
                max_vol := vol_in_row
                current_poc := (row_low + row_high) / 2
        
        if not na(current_poc)
            poc_price := current_poc
            poc_hi := current_poc + row_size / 2
            poc_lo := current_poc - row_size / 2
            
            // Calculate Delta % at POC
            float buy_vol = 0
            float sell_vol = 0
            for j = 0 to profile_lookback - 1
                if close[j] >= poc_lo and close[j] <= poc_hi
                    if close[j] > open[j]
                        buy_vol += volume[j]
                    else
                        sell_vol += volume[j]
            float total_poc_vol = buy_vol + sell_vol
            if total_poc_vol > 0
                poc_delta_pct := math.abs(buy_vol - sell_vol) / total_poc_vol * 100

// ===== TREND CALCULATIONS =====
ema_val = ta.ema(close, ema_len)
ema_slope = ema_val - ema_val[ema_slope_len]
ema_up = ema_slope >= min_slope
ema_dn = ema_slope <= -min_slope

// ===== ENTRY LOGIC =====
// New Entry: Breakout of POC range in direction of EMA trend and Delta imbalance
bool long_condition = false
bool short_condition = false

if in_session and strategy.position_size == 0
    // Check if cooldown is over
    bool cooldown_ok = true
    if cooldown_bars > 0
        int bars_since_entry = 0
        for i = 1 to cooldown_bars
            if strategy.position_size[i] != 0
                bars_since_entry := i
                break
        cooldown_ok = bars_since_entry < cooldown_bars ? false : true
        
    if cooldown_ok and not na(poc_price) and poc_delta_pct >= min_delta_pct
        // Long: Close breaks above POC High, EMA is up, Delta is positive (buy > sell)
        if close > poc_hi and ema_up and poc_delta_pct > 0
            // Verify delta direction: Buy volume > Sell volume at POC
            float buy_vol_check = 0
            float sell_vol_check = 0
            for j = 0 to profile_lookback - 1
                if close[j] >= poc_lo and close[j] <= poc_hi
                    if close[j] > open[j]
                        buy_vol_check += volume[j]
                    else
                        sell_vol_check += volume[j]
            if buy_vol_check > sell_vol_check
                long_condition := true
                
        // Short: Close breaks below POC Low, EMA is down, Delta is negative (sell > buy)
        if close < poc_lo and ema_dn and poc_delta_pct > 0
            // Verify delta direction: Sell volume > Buy volume at POC
            float buy_vol_check = 0
            float sell_vol_check = 0
            for j = 0 to profile_lookback - 1
                if close[j] >= poc_lo and close[j] <= poc_hi
                    if close[j] > open[j]
                        buy_vol_check += volume[j]
                    else
                        sell_vol_check += volume[j]
            if sell_vol_check > buy_vol_check
                short_condition := true

// ===== EXECUTION =====
var float peak_profit_pts = 0
var float current_stop = na

// desk indicator ta.rsi
_desk_gate = ta.rsi(close, 14)
_desk_long = _desk_gate > 50
_desk_short = _desk_gate < 50
if long_condition and _desk_long
    strategy.entry("L", strategy.long)
    current_stop := long_stop
    peak_profit_pts := 0
    
if short_condition and _desk_short
    strategy.entry("S", strategy.short)
    current_stop := short_stop
    peak_profit_pts := 0

// ===== BREAKEVEN LOGIC =====
if strategy.position_size > 0
    float avg = strategy.position_avg_price
    float profit_now = close - avg
    if use_breakeven and profit_now >= be_trigger_pts
        float new_be_stop = avg + 1 / pv // 1 tick above entry
        if current_stop < new_be_stop
            current_stop := new_be_stop
            strategy.exit("BE_L", "L", stop=long_stop, limit=long_limit)
            
if strategy.position_size < 0
    float avg = strategy.position_avg_price
    float profit_now = avg - close
    if use_breakeven and profit_now >= be_trigger_pts
        float new_be_stop = avg - 1 / pv // 1 tick below entry
        if current_stop > new_be_stop
            current_stop := new_be_stop
            strategy.exit("BE_S", "S", stop=long_stop, limit=long_limit)

// ===== EXIT LOGIC =====
if strategy.position_size > 0
    strategy.exit("XL", "L", stop=long_stop, limit=long_limit)
if strategy.position_size < 0
    strategy.exit("XS", "S", stop=long_stop, limit=long_limit)

if strategy.position_size == 0
    be_active := false

// ===== TRAILING STOP =====
if use_trail
    if strategy.position_size > 0
        float avg        = strategy.position_avg_price
        float profit_now = high - avg
        if profit_now > peak_profit_pts
            peak_profit_pts := profit_now
        if peak_profit_pts >= trail_min_dollars / pv
            float peak_dollars = peak_profit_pts * pv
            int   steps        = math.floor(peak_dollars / trail_step_amt)
            float keep_pct     = math.min(trail_pct + steps * trail_step_pct, 95.0)
            float new_stop     = avg + peak_profit_pts * (keep_pct / 100.0)
            if new_stop > current_stop
                current_stop := new_stop
                strategy.exit("XL", "Long", stop=long_stop, limit=long_limit)
    else if strategy.position_size < 0
        float avg        = strategy.position_avg_price
        float profit_now = avg - low
        if profit_now > peak_profit_pts
            peak_profit_pts := profit_now
        if peak_profit_pts >= trail_min_dollars / pv
            float peak_dollars = peak_profit_pts * pv
            int   steps        = math.floor(peak_dollars / trail_step_amt)
            float keep_pct     = math.min(trail_pct + steps * trail_step_pct, 95.0)
            float new_stop     = avg - peak_profit_pts * (keep_pct / 100.0)
            if new_stop < current_stop
                current_stop := new_stop
                strategy.exit("XS", "Short", stop=short_stop, limit=short_limit)

// ===== PLOT =====
plot(ema_val, "200 EMA", ema_up ? color.lime : ema_dn ? color.red : color.gray, linewidth=2)
plot(not na(poc_price) ? poc_price : na, "POC", color.yellow, style=plot.style_stepline, linewidth=1)
plot(not na(poc_hi) ? poc_hi : na, "POC High", #2962FF, style=plot.style_stepline, linewidth=2)
plot(not na(poc_lo) ? poc_lo : na, "POC Low", #2962FF, style=plot.style_stepline, linewidth=2)
plot(strategy.position_size != 0 ? current_stop : na, "Stop", color.fuchsia, style=plot.style_stepline, linewidth=2)

bgcolor(not na(poc_hi) and not na(poc_lo) and close >= poc_lo and close <= poc_hi ? color.new(#2962FF, 90) : na)

plotshape(long_condition, "Long Signal", shape.triangleup, location.belowbar, color.lime, size=size.small)
plotshape(short_condition, "Short Signal", shape.triangledown, location.abovebar, color.red, size=size.small)

plot(not na(poc_delta_pct) ? poc_delta_pct : na, "POC Delta %", color.orange, display=display.data_window)
