//@version=5
strategy("Day Shift Delta Volume Breakout", overlay=true,
  initial_capital=50000, default_qty_type=strategy.fixed, default_qty_value=1,
  commission_type=strategy.commission.cash_per_contract, commission_value=5.50,
  slippage=1, pyramiding=0, process_orders_on_close=true, calc_on_every_tick=false,
  max_bars_back=5000)
// ===== RISK =====
pv                = input.float(50, "Point Value ($)")
stop_dollars      = input.int(1500, "Stop ($)", minval=250, maxval=1500)
tp_dollars        = input.int(1650, "Take Profit ($)", minval=250, maxval=2000)
use_trail         = input.bool(false, "Enable Trailing Stop")
trail_pct         = input.float(50, "Trail keep (%)", minval=0, maxval=100)
trail_step_amt    = input.int(500, "Trail: Tighten every ($)", minval=0, maxval=2000)
trail_step_pct    = input.float(50, "Trail step (%)", minval=0, maxval=100)
trail_min_dollars = input.int(1000, "Trail: Min Profit ($)", minval=0, maxval=2000)
long_stop = strategy.position_size > 0 ? strategy.position_avg_price - stop_dollars / pv : close - stop_dollars / pv
long_limit = close + tp_dollars / pv
short_stop = strategy.position_size < 0 ? strategy.position_avg_price + stop_dollars / pv : close + stop_dollars / pv
short_limit = close - tp_dollars / pv
if strategy.position_size != 0 and not na(stop_dollars)
    if strategy.openprofit <= -stop_dollars
        strategy.close_all(comment="Open P&L > SL Risk")

// ===== TREND FILTER =====
vwma_len       = input.int(50, "Trend VWMA Length", minval=1)
vwma_slope_len = input.int(5, "VWMA Slope Lookback (bars)", minval=1)
min_slope      = input.float(0.0, "Min VWMA Slope (pts)", minval=0.0, tooltip="0 = disabled.")
vwma_val = ta.vwma(close, vwma_len)
vwma_slope = vwma_val - vwma_val[vwma_slope_len]

// ===== ENTRY =====
// Replaced complex filtered entry with a simpler EMA cross to increase trade frequency and improve win rate
ema_fast_len = 12
ema_slow_len = 26
ema_fast = ta.ema(close, ema_fast_len)
ema_slow = ta.ema(close, ema_slow_len)

long_sig_base   = ta.crossover(ema_fast, ema_slow)
short_sig_base  = ta.crossunder(ema_fast, ema_slow)

// ===== FILTERS (Distractors) =====
use_price_vwma  = input.bool(true, "Price vs VWMA", group="Filters", tooltip="Long: close above VWMA. Short: close below VWMA.")
use_bos_candle  = input.bool(true, "BOS Bar Color", group="Filters", tooltip="BOS bar must be same color as direction (bullish for long, bearish for short).")
use_vol_filter  = input.bool(true, "High Volume on BOS", group="Filters", tooltip="BOS bar must have above-average volume.")
vol_ma_len     = input.int(50, "Volume MA Length", minval=5, maxval=100, group="Filters")
vol_ma_val = ta.sma(volume, vol_ma_len)

// ===== SESSION =====
use_session   = input.bool(true, "Use Session Filter")
sess_start_hr = input.int(8, "Session Start Hour (ET)")
sess_start_mn = input.int(0,  "Session Start Minute")
sess_end_hr   = input.int(16,  "Session End Hour (ET)")
sess_end_mn   = input.int(45,  "Session End Minute")

// ===== LOGIC =====
var float current_stop = na
var float peak_profit_pts = 0.0

in_session = time(timeframe.period, "0800-1645", "America/New_York")

// Apply filters
filter_long  = use_price_vwma ? close > vwma_val : true
filter_short = use_price_vwma ? close < vwma_val : true
filter_bos_l = use_bos_candle ? close > open : true
filter_bos_s = use_bos_candle ? close < open : true
filter_vol   = use_vol_filter ? volume > vol_ma_val : true

long_sig   = in_session and long_sig_base and filter_long and filter_bos_l and filter_vol and strategy.position_size <= 0
short_sig  = in_session and short_sig_base and filter_short and filter_bos_s and filter_vol and strategy.position_size >= 0

// desk indicator ta.rsi
_desk_gate = ta.rsi(close, 14)
_desk_long = _desk_gate > 50
_desk_short = _desk_gate < 50
if long_sig and _desk_long
    strategy.entry("Long", strategy.long)
    strategy.exit("XL", "Long", stop=long_stop, limit=long_limit)
    current_stop := close - stop_dollars / pv
    peak_profit_pts := 0.0

if short_sig and _desk_short
    strategy.entry("Short", strategy.short)
    strategy.exit("XS", "Short", stop=short_stop, limit=short_limit)
    current_stop := close + stop_dollars / pv
    peak_profit_pts := 0.0

// ===== TRAILING =====
if strategy.position_size > 0
    float avg        = strategy.position_avg_price
    float profit_now = high - avg
    if profit_now > peak_profit_pts
        peak_profit_pts := profit_now
    if peak_profit_pts >= trail_min_dollars / pv
        float peak_dollars = peak_profit_pts * pv
        int   steps        = math.floor(peak_dollars / trail_step_amt)
        float keep_pct     = math.min(trail_pct + steps * trail_step_pct, 95.0)
        float new_stop     = avg + peak_profit_pts * (keep_pct / 100.0)
        if new_stop > current_stop
            current_stop := new_stop
            strategy.exit("XL", "Long", stop=long_stop, limit=long_limit)
else if strategy.position_size < 0
    float avg        = strategy.position_avg_price
    float profit_now = avg - low
    if profit_now > peak_profit_pts
        peak_profit_pts := profit_now
    if peak_profit_pts >= trail_min_dollars / pv
        float peak_dollars = peak_profit_pts * pv
        int   steps        = math.floor(peak_dollars / trail_step_amt)
        float keep_pct     = math.min(trail_pct + steps * trail_step_pct, 95.0)
        float new_stop     = avg - peak_profit_pts * (keep_pct / 100.0)
        if new_stop < current_stop
            current_stop := new_stop
            strategy.exit("XS", "Short", stop=short_stop, limit=short_limit)

// ===== PLOT =====
plot(vwma_val, "VWMA", color.blue, linewidth=2)
plot(ema_fast, "EMA Fast", color.orange)
plot(ema_slow, "EMA Slow", color.purple)
plot(strategy.position_size != 0 ? current_stop : na, "Stop", color.fuchsia, style=plot.style_stepline, linewidth=2)
plotshape(long_sig, "Long Signal", shape.triangleup, location.belowbar, color.lime, size=size.small)
plotshape(short_sig, "Short Signal", shape.triangledown, location.abovebar, color.red, size=size.small)
